# harness: fast-forward ok (Session 55 part 1, the red blocks: damage builds up on the wrestlers during a match)
# A launched Quick Match (e.g. --wwe2k14_launch=@quick --wwe2k14_quick_type=0 --wwe2k14_launch_participants=218,276
# --wwe2k14_fixed_seed=13 --wwe2k14_launch_skip_entrances=true): from the match's start a screenshot every 600 frames
# for 24,000 frames (NNN_d_01..40: the damage the game paints on skin builds up, foreheads and backs), or until the
# match is decided, then quit. Compare a run with the slow-compile emulation and the old behaviour against one with
# the fix. on_fail quit.
on_fail quit
include boot_to_menu
wait_file INGAMELUA.PAC 3600
wait_state seq 9 3600 30
wait_state match_clock >600 30000 2
log s55d: match running
screenshot_every 600 d
wait_state 0x8376ACCC !=0 24000 2 optional
screenshot_every 0
screenshot d_end
log s55d: done
quit
